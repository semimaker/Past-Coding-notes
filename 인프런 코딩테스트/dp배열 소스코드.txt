#include <iostream>
#include <algorithm>
#include <string.h> //#include <string>

using namespace std;

char s1[1010];
char s2[1010];

int len1; //문자열의 길이.
int len2;

int DP[1010][1010]; //불필요한 동적 할당을 만들지 않을 수 있다.

int main(){
	
	scanf("%s", &s1[1]); //dp배열을 채울 떄 0번째 인덱스를 채우는데에 필요.
	scanf("%s", &s2[1]);

	len1 = strlen(&s1[1]);
	len2 = strlen(&s2[1]);

	//---------------------solution-------------------   
	for (int i = 1; i <= len1; i++) {
		for (int j = 1; j <= len2; j++){
			//DP[i][j]을 채우기 위한 점화식
			//(i, j)
			if (s1[i] == s2[j])
				DP[i][j] = DP[i - 1][j - 1] + 1;
			else
				DP[i][j] = max(DP[i - 1][j], DP[i][j - 1]);
		}
	}

	printf("%d", DP[len1][len2]);

}
