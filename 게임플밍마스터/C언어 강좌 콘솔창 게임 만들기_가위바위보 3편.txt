#include <stdio.h>
#include <stdlib.h>
#include <time.h> //timer 함수 전용.

int g_result[3][3] =
{
	//가위 0, 바위 1, 보 2 com

	{0,-1,1}, //하나의 소스로서 만들었다.
	{1,0,-1},
	{-1,1,0}
	//가위 0 -> 0/-1/1(무/패/승)
	//바위 1
	//보   2
};//gloval underbar 

int main(void)
{
	int player, com, result = 0; //가위바위보를 위한 세 개의 정수, 가위/바위/보.

	srand(time(NULL)); 
	//반복된 계산식을 통해서, 하나의 임의의 숫자를 만듬
	//같은 값이 연달아 나올 것이 연달아 나오지 않게 하기 위해서,
	//1970년 1월 1일부터 현재까지의 초 단위 및 통계(시간)을 집어넣는다.
	int value = 0; //뺄값 정이

	while (1)
	{
		printf("가위(0), 바위(1), 보(2) 중 하나를 선택하시요. >> \n");
		scanf_s("%d", &player);
		com = rand() % 3; //ctrl + D(단축키: 하나의 문장을 통째로 복사)

		value = player - com;//값을 빼는 계산식 
		//p : 가위 0 가위 0 = 0(무)
		//           바위 0 = -1(패)
		//             보 0 = -2(승) 승 : 1, -2  패 : -1,2 무 : 0
		//p : 바위 1 가위 0 = 1(승)
		//           바위 1 = 0(무)
		//             보 2 = -1(패)
		//p : 보 2 가위0 = 2(패)
		//         바위 1 = 1(승)
		//           보 2 = 0(무)
	
		//p :가위 0
		if (player == 0)
		{
			if (com == 0) result = 0; //printf("비겼습니다. \n");
			else
			{
				if (com == 1) result = -1;
				else result = 1;
			}
		}

		// p : 바위 1 
		if (player == 1)
		{
			if (com == 0) result = 1;
			else
			{
				if (com == 1) result = 0;
				else result = -1;
			}
		}
		// p : 보 2
		if (player == 2)
		{
			if (com == 0) result = -1;
			else
			{
				if (com == 1) result = 1;
				else result = 0;
			}
		}
		result = g_result[player][com];
		value = result; //값을 처음부터 넣어버리면 라인을 줄여서 사용이 가능하다.

		if (value == 1 || value == -2) result = 1; // 승 : 1,2
		if (value == -1 || value == 2) result = -1; // 패 : -1,2
		if (value == 0) result = 0; // 무 : 0

		printf("player : %d vs com : %d = %d", player, com, result);
	}
	

	system("pause");
	return 0;
}
