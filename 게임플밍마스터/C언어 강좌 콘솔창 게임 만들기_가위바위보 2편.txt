#include <stdio.h>
#include <stdlib.h>
#include <time.h> //timer 함수 전용.
int main(void)
{
	int player, com, result = 0; //가위바위보를 위한 세 개의 정수, 가위/바위/보.

	srand(time(NULL)); 
	//반복된 계산식을 통해서, 하나의 임의의 숫자를 만듬
	//같은 값이 연달아 나올 것이 연달아 나오지 않게 하기 위해서,
	//1970년 1월 1일부터 현재까지의 초 단위 및 통계(시간)을 집어넣는다.

	while (1)
	{
		printf("가위(0), 바위(1), 보(2) 중 하나를 선택하시요. >> \n");
		scanf_s("%d", &player);
		com = rand() % 3; //ctrl + D(단축키: 하나의 문장을 통째로 복사)

		//p :가위 0
		if (player == 0)
		{
			if (com == 0) result = 0; //printf("비겼습니다. \n");
			else
			{
				if (com == 1) result = -1;
				else result = 1;
			}
		}

		// p : 바위 1 
		if (player == 1)
		{
			if (com == 0) result = 1;
			else
			{
				if (com == 1) result = 0;
				else result = -1;
			}
		}
		// p : 보 2
		if (player == 2)
		{
			if (com == 0) result = -1;
			else
			{
				if (com == 1) result = 1;
				else result = 0;
			}
		}

		printf("player : %d vs com : %d = %d", player, com, result);
	}
	

	system("pause");
	return 0;
}
