#include <stdio.h>//기본 라이브러리
#include <stdlib.h>
#include <time.h>//time 관련 라이브러리.
int main(void)
{
	int com;
	int player;
	int count = 0;
	//컴퓨터가 임의의 숫자를 결정. NULL은 0이라고 여기면 된다.

	srand(time(NULL));
	//(srand로 값을 받음)첫 값을 이용하여 srand 출력하며, 그 값이 빠져나오면 된다.
	com = rand() % 100 + 1;
	//임의의 숫자 결정. 1~100 사이 숫자 출력 시, 필요한 범위만큼의 나머지를 사용.
	//count++;//후입력의 형태로 값을 받음


	while (1) //원하는 답이 나올때까지 반복적으로 제시한다.
	{
		count++;//답이 틀릴때마다, while문에 의해 반복, 여기서 점점 값이 증가.
		printf("숫자를 입력하세요 (1 ~ 100)"); //플레이어는 숫자를 입력한다.
		scanf_s("%d", &player); //임의의 숫자보다 낮으면 "Low"라고 출력

		if (player < com) //임의의 숫자보다 낮으면(LOW) 
		{
			printf("Low \n");
		}

		else
		{
			if (player > com) //임의의 숫자보다 크면(HIGH)
			{
				printf("High \n");
			}
			else
			{
				printf("%d번만에 맞았습니다 \n", count);
				break;
			}
		}
	}
	system("pause"); //틀리든 말든 끝나버린다.
	return 0;
}
