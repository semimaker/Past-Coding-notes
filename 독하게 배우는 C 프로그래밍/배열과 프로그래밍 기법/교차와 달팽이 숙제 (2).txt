#include <stdio.h>

int main(void)
{
	int aList[5][5] = { 0 };
	int cnt = 0;
	
	for (int i = 0; i < 5; ++i)
	{
		for (int j = 0; j < 5; ++j)//0,2,4는 짝수 자리 취급. 1,3,5는 홀수 자리 취급.
		{//0~5까지 증가. 보수의 개념을 활용하여 뒤집어서 처리된다.
			if (i % 2 == 0)
				aList[i][j] = ++cnt;
			else
				aList[i][4 - j] = ++cnt;
			//j가 0에서부터 증가하여 4,3,2,1 순으로 감소하여 등장.
		}
	}

	for (int i = 0; i < 5; ++i)
	{
		for (int j = 0; j < 5; ++j)
			printf("%d\t", aList[i][j]);
		putchar('\n');
	}
	return 0;
}
