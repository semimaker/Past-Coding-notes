#include <stdio.h>

int main(void)
{
	int aList[5] = { 30, 40, 10, 50, 20 };
	
	for (int i = 1; i < 5; ++i)
	{
		if (aList[0] > aList[i])
		{
			int tmp = aList[0];
			aList[0] = aList[i];
			aList[i] = tmp;
		}
	}
	
	for(int i = 0; i < 5; ++i)
		printf("%d\t", aList[i]);
	putchar('\n');
	return 0;
}
