//for문을 이용하여 int[5] 배열에 저장된 값을 오름차순으로
// 정렬하는 프로그램을 작성. 버블정렬 알고리즘을 사용.
#include <stdio.h>

int main(void)
{
	int aList[5] = { 30, 40, 10, 50, 20 };

	for (int i = 0; i < 5; ++i)
		printf("%d\t", aList[i]);
	putchar('\n');

	for (int j = 0; j < 5; ++j)
	{
		for (int i = j + 1; i < 5; ++i)
		{
			if (aList[0] > aList[i])
			{
				int tmp = aList[0];
				aList[0] = aList[i];
				aList[i] = tmp;
			}
		}


		for (int i = 0; i < 5; ++i)
			printf("%d\t", aList[i]);
		putchar('\n');
	}
	return 0;
}
//버블 정렬 
//1 5 3 4 2 ->
//1 2 5 4 3 ->
//1 2 5 4 3 ->
//1 2 3 5 4 ->
//1 2 3 4 5
