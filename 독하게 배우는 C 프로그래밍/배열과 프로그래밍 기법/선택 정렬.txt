//for문을 이용하여 int[5] 배열에 저장된 값을 오름차순으로
// 정렬하는 프로그램을 작성. 버블정렬 알고리즘을 사용.
#include <stdio.h>

int main(void)
{
	int aList[5] = { 30, 40, 10, 50, 20 };
	int idxMin = 0;//가장 작은 값을 가진 것이 0번 요소이다.

	for (int j = 0; j < 4; ++j)
	{
		idxMin = j;
		for (int i = j + 1; i < 5; ++i)
		{
			if (aList[idxMin] > aList[i])//인덱스가 지정한 요소 > 리스트의 i번째 크기.
				idxMin = i;//idxMin을 업데이트한다.
		}

		if (idxMin != j) //더 작은 값을 찾은 경우.
		{
			int tmp = aList[j]; //기존의 j값을 인덱스에 대입하여 값을 출력한다.
			aList[j] = aList[idxMin];
			aList[idxMin] = tmp;
		}//교환을 할 거냐 말꺼냐.

		for (int i = 0; i < 5; ++i)
			printf("%d\t", aList[i]);
		putchar('\n');
	}
	
	return 0;

}
