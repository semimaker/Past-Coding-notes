#include <stdio.h>
int main(void) {

	int age = 0;
	double rate = 0.0;

	scanf_s("%d", &age);//입력되는 나이에 따라서 할인율이 적용
	if (age < 14)

		if (age < 4)
			rate = 0.0;
		else
			rate = 0.5;
	else
		if (age >= 20)
			rate = 1.0;
		else
			rate = 0.75;

	double fee = 1000 * rate;
	printf("Fee : %d", (int)fee);
	return 0;
} //경우의 수를 나누는 2분할 방식으로 하나씩 찾아내는 2분검색 방식을 사용.
