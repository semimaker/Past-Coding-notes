#include <stdio.h>
int main() {

	int a = 3;
	int b = 4;
	int tmp;

	tmp = a;
	a = b;
	b = tmp; //기존의 숫자를 바꾼 것이다.
	// 3 -> a -> tmp -> b, 4 -> b -> a 로 이동하면서 저장된 셈이다.

	printf("두 정수의 합은 a :%d, b: %d 입니다.\n", a, b);
	return 0;

}
