#include <stdio.h>

int main() {
	
	int a = 0, b = 0;
	int result = 0;
	scanf_s("%d %d", &a, &b);

	result = a + (~b + 1); 
	//result =  a - b, ~ : not 연산자.
	//2의 보수가 되는 형식.

	//result % 0x00000001; bit연산자가 bitwize연산자보다 효율이 높다.

	printf("%d\n", result);

	return 0;
}
//2의 보수 계산 
//9 => 1001, 6 => 0110. 3=> 0011.
//~b = ~6 = 1001(1의 보수, 각 자릿수 값이 1인 수에서 2진수를 제거.) + 1 = 1010 
//1001 + 1010 = 10011 => 맨 앞 1은 4bit에 의해 제거. 0011(2) = 3이 된다.
