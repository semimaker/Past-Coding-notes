#include <stdio.h>

int main(void)
{
	for (int i = 0; i < 5; ++i)
	{
		for (int j = 0; j < 5; ++j)
		{
			if(j >= 4 - i)//i을 빼버리는 방법. 처음에는 i=0이므로, 4가 되어서 전부 공백.
				printf("* ");//크거나 같으면 표시
			else
				printf("  ");//아니면 여백. 
			//괄호의 넓힘에 따라서 출력된 모양이 달라진다.
		
		}
		putchar('\n');
	}
	return 0;
}
