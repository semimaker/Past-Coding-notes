module uart_tx(
    reset,
    mclk,
    baudrate,
    parity_sel,
    stop_sel,
    tdata,
    send,
    txd,
    done
);
input reset;
input mclk;
input [15:0] baudrate;
input [1:0] parity_sel;
input stop_sel;
input [7:0] tdata;
input send;
output txd;
output done;
//여기까지는 port 선언.

reg m_state;
parameter M_IDLE = 1'b0;
parameter M_RUN = 1'b1;
//define state/IDLE/RUN , IDLE 상태에서 send flag가 작동하면 run 되면서 데이터를 전송. 
//전송이 완료되면 done flag을 하고 IDLE 모드가 된다. 
wire s_idle = (m_state == M_IDLE ) ? 1'b1 : 1'b0;
wire s_run = (m_state == M_RUN ) ? 1'b1 : 1'b0;
//state flag을 정의한다. 

reg [15:0] cnt1;
always @(posedge mclk or negedge reset)
begin
        if(~reset) cnt1 <= 16'b0;
        else       cnt1 <= s_idle ? 16'b0 : (cnt1 == baudrate) ? 16'b0 : cnt1+1'b1;
end
//baudrate까지 countimg.1 bit duration.
reg [15:0] cnt2;
always @(posedge mclk or negedge reset)
begin
        if(~reset) cnt2 <= 4'b0;
        else       cnt2 <= s_idle ? 4'b0 : (cnt1 == baudrate) ? cnt2+1'b1 : cnt2;
end
//전송하는 bit counter 정의.
wire tdata_xor = tdata[0]^tdata[1]^tdata[2]^tdata[3]^tdata[4]^tdata[5]^tdata[6]^tdata[7];
reg  parity;
always @(posedge mclk or negedge reset)
begin
        if(~reset) parity <= 1'b0;
        else       parity <= (parity_sel == 2'b01) ? tdata_xor : ~tdata_xor;
end//전송할 데이터의 parity bit을 계산. 
reg txd;
always @(posedge mclk or negedge reset)
begin
        if(~reset) txd <= 1'b1;
        else       txd <= s_idle ? 1'b1:
                        ((cnt2==4'd0) && (cnt1==16'd0)) ? 1'b0:
                        ((cnt2==4'd1) && (cnt1==16'd0)) ? tdata[0]:
                        ((cnt2==4'd2) && (cnt1==16'd0)) ? tdata[1]:
                        ((cnt2==4'd3) && (cnt1==16'd0)) ? tdata[2]:
                        ((cnt2==4'd4) && (cnt1==16'd0)) ? tdata[3]:
                        ((cnt2==4'd5) && (cnt1==16'd0)) ? tdata[4]:
                        ((cnt2==4'd6) && (cnt1==16'd0)) ? tdata[5]:
                        ((cnt2==4'd7) && (cnt1==16'd0)) ? tdata[6]:
                        ((cnt2==4'd8) && (cnt1==16'd0)) ? tdata[7]:
                        
                        ( (parity_sel==2'b0) & ~stop_sel & (cnt2==4'd9) & (cnt1==16'd0) ) ? 1'b1 :
                        
                        ( (parity_sel==2'b0) & stop_sel & (cnt2==4'd9) & (cnt1==16'd0) ) ? 1'b1 :
                        ( (parity_sel==2'b0) & stop_sel & (cnt2==4'd10) & (cnt1==16'd0) ) ? 1'b1 :
                        
                        ( (parity_sel!=2'b0) & ~stop_sel & (cnt2==4'd9) & (cnt1==16'd0) ) ? parity :
                        ( (parity_sel!=2'b0) & ~stop_sel & (cnt2==4'd10) & (cnt1==16'd0) ) ? 1'b1 :
                        
                        ( (parity_sel!=2'b0) & stop_sel & (cnt2==4'd9) & (cnt1==16'd0) ) ? parity:
                        ( (parity_sel!=2'b0) & stop_sel & (cnt2==4'd10) & (cnt1==16'd0) ) ? 1'b1 : 
                        ( (parity_sel!=2'b0) & stop_sel & (cnt2==4'd11) & (cnt1==16'd0) ) ? 1'b1 :
                        txd ;
end
//전송되는 비트 txd을 생성한다. 
wire done = s_idle; //done을 생성한다.    

always @(posedge mclk or negedge reset)
begin
           if(~reset)  m_state <= 1'b0;
           else        m_state <= (s_idle & send) ? M_RUN : 
                                   
                                   (s_run & (parity_sel==2'b0) & (stop_sel==1'b0) & (cnt2==4'd10)) ? M_IDLE :
                                   (s_run & (parity_sel==2'b0) & (stop_sel==1'b1) & (cnt2==4'd11)) ? M_IDLE :
                                   (s_run & (parity_sel==2'b0) & (stop_sel==1'b0) & (cnt2==4'd11)) ? M_IDLE :
                                   (s_run & (parity_sel==2'b0) & (stop_sel==1'b1) & (cnt2==4'd12)) ? M_IDLE :
                                   m_state;
end      
//상태 전이를 구현합니다. 
endmodule

`timescale 1ns / 1ps
//////////////////////////////////////////////////////////////////////////////////
// Company: 
// Engineer: 
// 
// Create Date: 2023/06/23 10:46:36
// Design Name: 
// Module Name: uart_rx
// Project Name: 
// Target Devices:

module uart_rx(
    reset,
    mclk,
    baudrate,
    parity_sel,
    stop_sel,
    ren,
    rdata,
    rvaild,
    overrun,
    frame_err,
    parity_err,
    rxd
    );
input reset;
input mclk;
input [15:0] baudrate;
input [1:0] parity_sel;
input stop_sel;
input ren;
output [7:0] rdata;
output rvaild;
output overrun;
output frame_err;
output parity_err;
input rxd;

reg [1:0] m_state;
parameter M_IDLE = 2'b0;
parameter M_RECEIVE = 2'd1;
parameter M_DONE = 2'd2;
//state 정의 .
wire s_idle = (m_state == M_IDLE ) ? 1'b1 : 1'b0;
wire s_receive = (m_state == M_RECEIVE) ? 1'b1 : 1'b0 ;
wire s_done = (m_state == M_DONE ) ? 1'b1 : 1'b0;

reg rxd_1d, rxd_2d, rxd_3d;
wire rxd_nedge = ~rxd_2d & rxd_3d;
always @(posedge mclk or negedge reset)
begin
        if(~reset) begin
            rxd_1d <= rxd;
            rxd_2d <= rxd_1d;
            rxd_3d <= rxd_2d;
        end
end

reg [15:0] cnt1;
always @(posedge mclk or negedge reset)
begin
    if(~reset) cnt1 <= 16'b0;
    else       cnt1 <= ~s_receive ? 16'b0 : (cnt1==baudrate) ? 16'b0 : cnt1+1'b1;
end

reg [3:0] cnt2;
always @(posedge mclk or negedge reset)
begin
    if(~reset) cnt2 <= 4'b0;
    else cnt2 <= ~s_receive ? 4'b0 : (cnt1 == baudrate) ? cnt2+1'b1 : cnt2;
end

reg[7:0] rxd_data;
always @(posedge mclk or negedge reset)
begin
        if(~reset) rxd_data <= 8'b0;
        else begin
            rxd_data[0] <= ((cnt2== 4'd1) & (cnt1 == {1'b0, baudrate[15:1]})) ? rxd_3d : rxd_data[0];
            rxd_data[1] <= ((cnt2== 4'd2) & (cnt1 == {1'b0, baudrate[15:1]})) ? rxd_3d : rxd_data[1];
            rxd_data[2] <= ((cnt2== 4'd3) & (cnt1 == {1'b0, baudrate[15:1]})) ? rxd_3d : rxd_data[2];
            rxd_data[3] <= ((cnt2== 4'd4) & (cnt1 == {1'b0, baudrate[15:1]})) ? rxd_3d : rxd_data[3];
            rxd_data[4] <= ((cnt2== 4'd5) & (cnt1 == {1'b0, baudrate[15:1]})) ? rxd_3d : rxd_data[4];
            rxd_data[5] <= ((cnt2== 4'd6) & (cnt1 == {1'b0, baudrate[15:1]})) ? rxd_3d : rxd_data[5];
            rxd_data[6] <= ((cnt2== 4'd7) & (cnt1 == {1'b0, baudrate[15:1]})) ? rxd_3d : rxd_data[6];
            rxd_data[7] <= ((cnt2== 4'd8) & (cnt1 == {1'b0, baudrate[15:1]})) ? rxd_3d : rxd_data[7];
         end
end

wire cal_parity = rxd_data[0]^rxd_data[1]^rxd_data[2]^rxd_data[3]^rxd_data[4]^rxd_data[5]^rxd_data[6]^rxd_data[7];

reg cal_parity2;
always @(posedge mclk or negedge reset)
begin
        if(~reset) cal_parity2 <=1'b0;
        else       cal_parity2 <= (parity_sel==2'b01) ? cal_parity : (parity_sel==2'b10) ? ~cal_parity : 1'b0;
        
end

reg rxd_parity;
always @(posedge mclk or negedge reset)
begin
        if(~reset) rxd_parity <=1'b0;
        else       rxd_parity <= ((cnt2==4'd9) & (cnt1=={1'b0, baudrate[15:1]})) ? rxd_3d :rxd_parity;
end

reg stop_bit1;
always @(posedge mclk or negedge reset)
begin
        if(~reset) stop_bit1 <=1'b0;
        else       stop_bit1 <= ( (parity_sel==2'b000) & (cnt2==4'd9) & (cnt1 == {1'b0, baudrate[15:1]}) ) ? rxd_3d :
                                ( (parity_sel==2'b000) & (cnt2==4'd9) & (cnt1 == {1'b0, baudrate[15:1]}) ) ? rxd_3d :
                                stop_bit1; 
end                                

reg stop_bit2;
always @(posedge mclk or negedge reset)
begin
        if(~reset) stop_bit2 <=1'b0;
        else       stop_bit2 <= ( (parity_sel==2'b000) & (cnt2==4'd9) & (cnt1 == {1'b0, baudrate[15:1]}) ) ? rxd_3d :
                                ( (parity_sel==2'b000) & (cnt2==4'd9) & (cnt1 == {1'b0, baudrate[15:1]}) ) ? rxd_3d :
                                stop_bit2; 
end                                

reg [2:0] cnt_done;
always @(posedge mclk or negedge reset)
begin 
    if(~reset) cnt_done <=3'b0;
    else       cnt_done <= ~s_done ? 3'b0 : cnt_done+1'b1;
end

always @(posedge mclk or negedge reset)
begin
    if(~reset) m_state <= 1'b0;
    else       m_state <= ( s_idle & rxd_nedge              ) ? M_RECEIVE : 
                ( (parity_sel==2'b0) & (stop_sel== 1'b0) & (cnt2==4'd9) & (cnt1==baudrate) ) ? M_DONE:
               ( (parity_sel==2'b0) & (stop_sel== 1'b1) & (cnt2==4'd10) & (cnt1==baudrate) ) ? M_DONE:
               ( (parity_sel==2'b0) & (stop_sel== 1'b0) & (cnt2==4'd10) & (cnt1==baudrate) ) ? M_DONE:
                ( (parity_sel==2'b0) & (stop_sel== 1'b1) & (cnt2==4'd11) & (cnt1==baudrate) ) ? M_DONE:
                (cnt_done == 3'd7 ) ? M_IDLE:
                 m_state;
 end
 
reg frame_err;
always @(posedge mclk or negedge reset)
begin
    if(~reset) frame_err <= 1'b0;
    else       frame_err <= ( (cnt_done==3'd5) & (stop_sel ==1'b0) ) ? ~stop_bit1 :
                            ( (cnt_done==3'd5) & (stop_sel ==1'b0) ) ? ~(stop_bit1&stop_bit2) :
                             frame_err;
end

reg parity_err;
always @(posedge mclk or negedge reset)
begin
    if(~reset) parity_err <= 1'b0;
    else       parity_err <=(parity_sel==2'b0) ? 1'b0 : 
                            ( (cnt_done==3'd5) & (cal_parity2==rxd_parity )) ? 1'b00 :
                            ( (cnt_done==3'd5) &  (cal_parity2!=rxd_parity )) ? 1'b01 :
                             parity_err;
end

wire fifo_full;
wire overrun = fifo_full;

wire fifo_empty;
wire rvaild = ~fifo_empty;

wire [7:0] fifo_din =rxd_data;
wire       fifo_wen = (cnt_done==3'd2) ? 1'b1 : 1'b0;
wire       fifo_ren = ren;
wire [7:0] fifo_dout;
wire [7:0] rdata = fifo_dout;

/*fifo_16x8    rxd_fifo (

    .rd_en(fifo_ren),
    .dout(fifo_dout),
    ,full(fifo_full),
    .empty(fifo_empty),
    .vaild(        )
);
*/
endmodule

`timescale 1ns / 1ps
//////////////////////////////////////////////////////////////////////////////////
// Company:

module i2c_master(
    reset,
    mclk,
    period,
    scl_o,
    sda_i,
    sda_o,
    data_w,
    data_r,
    cmd,
    status
);
 input   reset;
 input   mclk;
 input [7:0]   period;
 output   scl_o;
 input   sda_i;
 output   sda_o;
 input [7:0]   data_w;
 output [7:0]   data_r;
 input [4:0]   cmd;
 output [1:0]   status;
//여기까지가 port 선언. 
wire [9:0] period2 = {period, 2'b0} ; // 통신 속도 설정. 8 -> 10bit.

wire send = cmd[4];//send flag
parameter SDAI_POS1 = 10'd50; //SDA 데이터를 읽음. 
parameter SDAO_POS = 10'd4;
//SDA data transition이 발생. 
reg [2:0] m_state;
parameter M_IDLE =       3'd0;
parameter M_START_RUNW = 3'd1;
parameter M_RUNW =       3'd2;
parameter M_STOP_RUNW =  3'd3;
parameter M_START_RUNR = 3'd4;
parameter M_RUNR =       3'd5;
parameter M_STOP_RUNR =  3'd6;

wire s_idle = (m_state == M_IDLE ) ? 1'b1 : 1'b0;
wire s_start_runw = (m_state == M_START_RUNW ) ? 1'b1 : 1'b0;
wire s_runw = (m_state == M_RUNW ) ? 1'b1 : 1'b0;
wire s_stop_runw = (m_state == M_STOP_RUNW ) ? 1'b1 : 1'b0;
wire s_start_runr = (m_state == M_START_RUNR ) ? 1'b1 : 1'b0;
wire s_runr = (m_state == M_RUNR ) ? 1'b1 : 1'b0;
wire s_stop_runr = (m_state == M_STOP_RUNR ) ? 1'b1 : 1'b0;
//define stat & state flag
reg [9:0] srw_cnt1;
always @(posedge mclk or negedge reset)
begin
        if(~reset) srw_cnt1 <= 10'b0;
        else       srw_cnt1 <= ~s_start_runw ? 10'b0 : send ? 10'b0 : 
                            (srw_cnt1==period2) ? 10'b0 : srw_cnt1+1'b1;
                             
end

reg [4:0] srw_cnt2;
always @(posedge mclk or negedge reset)
begin
       if(~reset) srw_cnt2 <= 5'b0;
       else       srw_cnt2 <= ~s_start_runw ? 5'b0 : send ? 5'b0 : 
                            (srw_cnt1 == period2) ? ((srw_cnt2==5'd20) ? 5'd20 :  srw_cnt2+1'b1) : srw_cnt2;
end 

reg srw_done;
always @(posedge mclk or negedge reset)
begin
    if(~reset) srw_done <= 1'b0;
    else       srw_done <= send ? 1'b0 : ( (srw_cnt2== 5'd20) && (srw_cnt1==SDAO_POS) ) ? 1'b1 : srw_done;

end

reg [9:0] rw_cnt1;
always @(posedge mclk or negedge reset)
begin
        if(~reset) rw_cnt1 <= 10'b0;
        else       rw_cnt1 <= ~s_start_runw ? 10'b0 : send ? 10'b0 : 
                            (rw_cnt1==period2) ? 10'b0 : rw_cnt1+1'b1;
                             
end

reg [4:0] rw_cnt2;
