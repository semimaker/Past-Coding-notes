`timescale 1ns / 1ps
module Sorting(clk, reset, en, a, b, FND, FNDSel4, FNDSel3, FNDSel2, FNDSel1);
	input			clk, reset, en;
	input [3:0]		a, b;
	output[6:0]		FND;
	output 			FNDSel4, FNDSel3, FNDSel2, FNDSel1;

	reg [6:0]		FND;
	reg 			FNDSel4, FNDSel3, FNDSel2, FNDSel1;
	
//	reg			clk2, clk100Hz;
	reg [6:0]		FNDa, FNDb, FNDc, FNDd;
	reg [3:0]	Greater, Less;
//	integer			m=0;

	reg	[25:0]	cnt ;
	always @(posedge clk or negedge reset)
	begin
		if(~reset)	cnt <= 26'd0;
	else		cnt <= (cnt==26'd5000) ? 26'd0 : cnt+1'b1;
	end

	reg	[3:0]	clock ;
	always @(posedge clk or negedge reset)
	begin
		if(~reset)	clock <= 4'b0;
	else		clock <= (cnt==26'd5000) ? clock + 1'b1 : clock ;
end

	always @ (posedge clk)
		if(en==1'b1)
			if (a > b) begin
				Greater <= a;
				Less <= b;
			end else begin
				Greater <= b;
				Less <= a;
			end
	
	always @(posedge clk) begin
		case(a)
			4'b0000 : FNDa = 7'b1000000; //0
			4'b0001 : FNDa = 7'b1111001; //1
			4'b0010 : FNDa = 7'b0100100; //2
			4'b0011 : FNDa = 7'b0110000; //3
			4'b0100 : FNDa = 7'b0011001; //4
			4'b0101 : FNDa = 7'b0010010; //5
			4'b0110 : FNDa = 7'b00000010; //6
			4'b0111 : FNDa = 7'b1011000; //7
			4'b1000 : FNDa = 7'b00000000; //8
			4'b1001 : FNDa = 7'b00010000; //9
			4'b1010 : FNDa = 7'b0100000; //10
			4'b1011 : FNDa = 7'b0000011; //11
			4'b1100 : FNDa = 7'b0100111; //12
			4'b1101 : FNDa = 7'b0100001; //13
			4'b1110 : FNDa = 7'b0000100; //14
			4'b1111 : FNDa = 7'b0001110; //15

		endcase

		case(b)
			4'b0000 : FNDb = 7'b1000000; //0
			4'b0001 : FNDb = 7'b1111001; //1
			4'b0010 : FNDb = 7'b0100100; //2
			4'b0011 : FNDb = 7'b0110000; //3
			4'b0100 : FNDb = 7'b0011001; //4
			4'b0101 : FNDb = 7'b0010010; //5
			4'b0110 : FNDb = 7'b00000010; //6
			4'b0111 : FNDb = 7'b1011000; //7
			4'b1000 : FNDb = 7'b00000000; //8
			4'b1001 : FNDb = 7'b00010000; //9
			4'b1010 : FNDb = 7'b0100000; //10
			4'b1011 : FNDb = 7'b0000011; //11
			4'b1100 : FNDb = 7'b0100111; //12
			4'b1101 : FNDb = 7'b0100001; //13
			4'b1110 : FNDb = 7'b0000100; //14
			4'b1111 : FNDb = 7'b0001110; //15
		endcase
		
			case(Less)
			4'b0000 : FNDc = 7'b1000000; //0
			4'b0001 : FNDc = 7'b1111001; //1
			4'b0010 : FNDc = 7'b0100100; //2
			4'b0011 : FNDc = 7'b0110000; //3
			4'b0100 : FNDc = 7'b0011001; //4
			4'b0101 : FNDc = 7'b0010010; //5
			4'b0110 : FNDc = 7'b00000010; //6
			4'b0111 : FNDc = 7'b1011000; //7
			4'b1000 : FNDc = 7'b00000000; //8
			4'b1001 : FNDc = 7'b00010000; //9
			4'b1010 : FNDc = 7'b0100000; //10
			4'b1011 : FNDc = 7'b0000011; //11
			4'b1100 : FNDc = 7'b0100111; //12
			4'b1101 : FNDc = 7'b0100001; //13
			4'b1110 : FNDc = 7'b0000100; //14
			4'b1111 : FNDc = 7'b0001110; //15
		endcase
		
		case(Greater)
			4'b0000 : FNDd = 7'b1000000; //0
			4'b0001 : FNDd = 7'b1111001; //1
			4'b0010 : FNDd = 7'b0100100; //2
			4'b0011 : FNDd = 7'b0110000; //3
			4'b0100 : FNDd = 7'b0011001; //4
			4'b0101 : FNDd = 7'b0010010; //5
			4'b0110 : FNDd = 7'b00000010; //6
			4'b0111 : FNDd = 7'b1011000; //7
			4'b1000 : FNDd = 7'b00000000; //8
			4'b1001 : FNDd = 7'b00010000; //9
			4'b1010 : FNDd = 7'b0100000; //10
			4'b1011 : FNDd = 7'b0000011; //11
			4'b1100 : FNDd = 7'b0100111; //12
			4'b1101 : FNDd = 7'b0100001; //13
			4'b1110 : FNDd = 7'b0000100; //14
			4'b1111 : FNDd = 7'b0001110; //15
		endcase
		
	end
	
	always @(clock,  FNDa, FNDb, FNDc, FND)
	begin 
	

	    if(reset & clock[0]) begin
			FNDSel1 = 1'b0;
			FNDSel2 = 1'b1;
			FNDSel3 = 1'b1;
			FNDSel4 = 1'b1;
			FND = FNDa;
	
		end else if (reset & clock[1]) begin
			FNDSel1 = 1'b1;
			FNDSel2 = 1'b0;
			FNDSel3 = 1'b1;
			FNDSel4 = 1'b1;
			FND = FNDb;

		end else if (en & clock[2]) begin	
			FNDSel1 = 1'b1;
			FNDSel2 = 1'b1;
			FNDSel3 = 1'b0;
 	   		FNDSel4 = 1'b1;
			FND = FNDc;
			//

		end else if (en & clock[3]) begin	
			FNDSel1 = 1'b1;
			FNDSel2 = 1'b1;
			FNDSel3 = 1'b1;
 	   		FNDSel4 = 1'b0;
			FND = FNDd;
			// 
	
		end else begin	
			FNDSel1 = 1'b1;
			FNDSel2 = 1'b1;
			FNDSel3 = 1'b1;
 	   		FNDSel4 = 1'b1;

			// 
		end
	end	
endmodule
