using System;
using System.Net;
using System.Net.Sockets;
using System.Text;
using System.Threading;

public class AutonomousCarController
{
    // 스마트폰 애플리케이션과 통신하기 위한 소켓 객체
    private Socket socket;
    private const int port = 8080;  // 사용할 포트 번호

    // 자동차 제어에 필요한 변수들
    private bool isMovingForward;
    private bool isMovingBackward;
    private bool isTurningLeft;
    private bool isTurningRight;

    public AutonomousCarController()
    {
        // 소켓 초기화
        socket = new Socket(AddressFamily.InterNetwork, SocketType.Stream, ProtocolType.Tcp);
    }

    public void ConnectToSmartphone(string ipAddress)
    {
        // 스마트폰 애플리케이션과 소켓 연결
        try
        {
            IPAddress serverIP = IPAddress.Parse(ipAddress);
            IPEndPoint serverEndPoint = new IPEndPoint(serverIP, port);
            socket.Connect(serverEndPoint);
            Console.WriteLine("Connected to smartphone at: " + socket.RemoteEndPoint);
        }
        catch (Exception e)
        {
            Console.WriteLine("Failed to connect to smartphone: " + e.Message);
        }
    }

    public void StartListening()
    {
        // 스마트폰으로부터 메시지를 수신하는 쓰레드 시작
        Thread receiveThread = new Thread(ReceiveMessages);
        receiveThread.Start();
    }

    private void ReceiveMessages()
    {
        while (true)
        {
            try
            {
                // 소켓으로부터 메시지 수신
                byte[] buffer = new byte[1024];
                int bytesRead = socket.Receive(buffer);
                string message = Encoding.UTF8.GetString(buffer, 0, bytesRead);
                Console.WriteLine("Received message: " + message);

                // 메시지 처리
                ProcessMessage(message);
            }
            catch (Exception e)
            {
                Console.WriteLine("Failed to receive message: " + e.Message);
                break;
            }
        }
    }

    private void ProcessMessage(string message)
    {
        // 메시지를 기반으로 자동차 동작 제어
        if (message.Contains("FORWARD"))
        {
            isMovingForward = true;
            MoveForward();
        }
        else if (message.Contains("BACKWARD"))
        {
            isMovingBackward = true;
            MoveBackward();
        }
        else if (message.Contains("LEFT"))
        {
            isTurningLeft = true;
            TurnLeft();
        }
        else if (message.Contains("RIGHT"))
        {
            isTurningRight = true;
            TurnRight();
        }
        else if (message.Contains("STOP"))
        {
            Stop();
        }
    }

    private void MoveForward()
    {
        // 자동차를 전진시키는 동작
        // 구현 내용을 여기에 추가하세요.
        Console.WriteLine("Moving forward");
    }

    private void MoveBackward()
    {
        // 자동차를 후진시키는 동작
        // 구현 내용을 여기에 추가하세요.
        Console.WriteLine("Moving backward");
    }

    private void TurnLeft()
    {
        // 자동차를 왼쪽으로 회전시키는 동작
        // 구현 내용을 여기에 추가하세요.
        Console.WriteLine("Turning left");
    }

    private void TurnRight()
    {
        // 자동차를 오른쪽으로 회전시키는 동작
        // 구현 내용을 여기에 추가하세요.
        Console.WriteLine("Turning right");
    }

    private void Stop()
    {
        // 자동차를 정지시키는 동작
        // 구현 내용을 여기에 추가하세요.
        Console.WriteLine("Stopping");
    }
}
이 코드는 C#을 사용하여 간단한 자동차 동작을 제어하기 위한 기본적인 구조를 제공합니다. 스마트폰 애플리케이션과의 통신은 소켓을 사용하며, ConnectToSmartphone 메서드로 스마트폰과 연결하고, StartListening 메서드로 스마트폰으로부터 메시지를 수신합니다. 메시지에 따라 ProcessMessage 메서드에서 자동차의 동작을 제어합니다.

각 동작(전진, 후진, 좌회전, 우회전, 정지)은 해당하는 메서드에서 구현해야 합니다. 현재는 각 메서드에는 동작을 실행하는 대신 메시지를 출력하도록 되어 있습니다. 실제 자동차 동작을 구현하기 위해 필요한 로직을 추가하세요.
