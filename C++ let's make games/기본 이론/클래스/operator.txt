#include <iostream>
#include <functional>

using namespace std;
/*
함수 포인터 : 메모리 주소를 저장하기 위한 포인터 변수 선언이 가능
전역함수의 경우, 함수명 = 함수의 메모리 구조
선언방법 - 반환타입(*포인터 변수명)(인자타입); 의 형태
*/
int Sum(int a, int b)
{
	return a + b;
}
int OutSum(int a, int b)
{
	cout << a - b << endl;
	return a - b;
}
float TestFunc(float a)
{
	cout << a << endl;
	return a;
}


void Output() {
	cout << "Output Function" << endl;
}

class CHanzo
{
public:
	CHanzo()
	{
		m_iTest = 10;
	}
	~CHanzo()
	{

	}
public:
	int m_iTest;

public:
	void Output()
	{
		cout << "Hanzo" << endl;

		cout << "Test : " << this->m_iTest << endl;//this는 생략가능.
	}
};

typedef struct _tagPoint
{
	int x;
	int y;

	_tagPoint() :
		x(0),
		y(0)
	{

	}
	_tagPoint(int _x, int _y) :
		x(_x),
		y(_y)
	{

	}
	_tagPoint(const _tagPoint& pt)
	{
		*this = pt;
	}//얕은 복사를 한다. 이렇게 함으로서 this는 자기자신의 포인터이고, *을 붙여 스스로를
	//역참조하여 모든 데이터를 복사한다.
	
	_tagPoint operator +(const _tagPoint& pt)
	{
		_tagPoint result;
		result.x = x + pt.x;
		result.y = y + pt.y;
		return result;
	};

	void operator << (const _tagPoint& pt)
	{
		x = pt.x;
		y = pt.y;
	}


_tagPoint operator +(int a)
{
	_tagPoint pt;
	pt.x = x + a;
	pt.y = y + a;
	return pt;
};

void operator +=(int a)
{
	x += a;
	y += a;
}
void operator ++()
{
	x++;
	y++;
}
void operator --()
{
	x--;
	y--;
}
void operator << (const _tagPoint& pt)
{
	x = pt.x;
	y = pt.y;
}
void operator >> (_tagPoint& pt)
{
	pt.x = x;
	pt.y = y;
}
}POINT, * PPOINT;

int main() {

	int(*pFunc)(int, int) = Sum;
	cout << pFunc(10, 20) << endl;

	pFunc = OutSum;
	OutSum(10, 20);

	void(*pFunc1)() = Output;
	pFunc1();

	CHanzo hanzo1, hanzo2;

	hanzo1.m_iTest = 100;
	hanzo2.m_iTest = 100;

	hanzo1.Output();
	hanzo2.Output();
	//this 포인터 : 클래스 안에서 this을 사용한다면 해당 객체의 메모리 주소가 된다.
	//즉, 자기 자신의 포인터이다. 
	//멤버함수를 호출 할 때, this가 호출자가 된다.

	void(CHanzo:: * pFunc2)() = &CHanzo::Output;
	//(hanzo1.*pFunc2)();
	(hanzo1.*pFunc2)();

	function<void()> func;
	function<void()> func1;

	func = bind(Output);
	func1 = bind(&CHanzo::Output, &hanzo2);

	cout << "================= Functional ==================" << endl;
	func();
	func1();

	function<int(int, int)> func3;
	function<float(float)> func4;
	//함수에 인자가 있을 경우, placeholders을 이용하여 인자의 순서를 정의할 수 있다.
	//아래처럼 2,1로 넣어주면 인자가 서로 바뀌게 된다.
	func3 = bind(OutSum, placeholders::_2, placeholders::_1);
	//위에서 2,1으로 바꿔쭸으므로, 10은 bdp 20은 a에 들어가게 된다.

	/*
	function기능은 C++11 부터 지원. 이 기능은 함수포인터를 전역, 멤버 가리지 않고
	쉽게 주소를 저장하여 호출 할 수 있게 만들어준다
	선언방법 : function<반환타입(인자타입)> 변수명;
	*/
	func3(10, 20);
	func4 = bind(TestFunc, placeholders::_1);
	func4(3.14f);

	cout << "================= Point ==================" << endl;
	POINT pt1(10, 20), pt2(30, 40), pt3;

	pt3 = pt1 + pt2;//point 구조체는 +연산자가 operater로 재정의.
	//그래서 + 연산이 가능하게 되고, pt1의 + 연산자 함수를 호출하는 개념.
	//pt2을 인자로 넘겨준다.

	pt3 << pt1;
	
	pt2 >> pt3;

	pt3 = pt1 + 1000;

	pt3 += 2000;

	++pt3;

	--pt3;

	cout << "x : " << pt3.x << "ty : " << pt3.y << endl;

	return 0;
}
