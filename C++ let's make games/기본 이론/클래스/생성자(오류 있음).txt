#include <iostream>

using namespace std;

//소멸자 : 어떤 클래스를 사용하여 생긴 객체가 메모리에서 해제될 때  
class CTracer
{
public: 
	CTracer()
	{
		cout << "Tracer 생성자" << endl;
		strcpy_s(m_strName, "트레이서");
	}//생성자 
	
	CTracer(char* pName)
	{
		strcpy_s(m_strName, pName);
		cout << "Name 생성자" << endl;
	}

	CTracer(char* pName, int iFlash, int iBack) :
		m_iBack(iBack),
		m_iFlash(iFlash)
	{
		strcpy_s(m_strName, pName);
		//m_iFlash = iFlash;
		cout << "Name, Flash, Back 생성자 " << endl;
		//int a;
		//a = 10;
		//int b = 10;

	}
	~CTracer()
	{
		cout << "Tracer 소멸자" << endl;
	}
	char m_strName[32]; //클래스 멤버변수 선언시 m_을 붙여준다.
public:
	int m_iAttack;
	int m_iHP;

private:
	int m_iFlash;
private:
	int m_iBack;
public:
	void Output()
	{
		cout << "Tracer Output" << endl;
		cout << "Name :" <<m_strName << endl;
	}
};

int main() {
	CTracer tr1("한조",3,1);
	tr1.Output();
	/*
	객체(Object) = 모든 사물, C++에서는 각 변수들도 해당한다. 하지만 OOP을 지원하기 위해
	제공하는 것은 class(!=객체)는 객체를 만들기 위한 틀.
	객체지향 프로그래밍(OOP) : 객체와의 관계를 설정하여 부품을 조립하듯이 객체들을 조립하여
	완성된 프로그램을 만들어나가는 방식.

	클래스 선언 방법 : class 클래스명 {}; 의 형태로 선언.
	코드블럭안에 멤버변수를 넣어줄 수 있다. 단, 함수도 멤버로 만들 수 있다.

	클래스의 4가지 속성 
	1. 캡슐화 -  클래스 안에 속하는 여러 변수, 함수를 하나의 클래스로 묶어주는 기능.
	2. 은닉화 -  클래스 안에 속하는 변수 혹은 내가 원하는 부분만 외부에 공개하고, 외부에 공개
	하지 않아도 된다. public, protected, private
	public : 클래스 내부, 외부에 모두 접근하는 방식.
	protected : 클래스 외부에서는 접근이 불가능하고 자기 자신 내부나 자식 클래스 내부에서는
	접근이 가능한 키워드.
	private : 자기자신 내부에서만 접근이 가능, 외부/자식 내부에서는 접근이 불가능한 영역.
	3. 상속성 -  클래스는 클래스-클래스 간 부모-자식 관계 형성.자식 클래스는 부모 클래스를
	물려받아 자기 것처럼 사용
	4. 다형성 - 부모 자식 관계로 상속관계가 형성된 클래스 간 서로 형변환이 가능한 성질.
	*/

	return 0;
}
