/*조건식이 true이면 코드블럭의 코드가 동작된다.
		무한루프 시에는 for(;;) {}으로 실행.
		for문에서 초기값은 처음 for문에 진입할 떄 딱 1번만 동작된다. 그 후, 조건식을 체크하고 조건식이 true이면,
		코드블럭의 코드가 동작한 후 증감값을 처리한다.
		그 후, 다시 조건을 체크하고 true이면 동작하고 증감 -> 조건 -> 증감 -> 조건의 순으로 처리.
		조건이 false 이거나 break을 만나면 for문을 빠져나간다.
		// 초기값 i = 0, 조건식 : i < 10 , 증감값 : ++i
		*/
		for (int i = 0; i < 10; ++i)
			// 먼저 for문에 진입하면서 초기값이 실행, i=0(초기화)
			//그 후, i<10을 체크하여 0이면 true 출력
			//증감을 처리하면서 ++을 하여 0 -> 1하고 조건 체크
			// 1은 10보다 작아서 true가 나오도록, 1출력 ++gotj 2가 되고 조건이 true가 된다
			// 2 출력 3,4,5,6,7,8,9까지는 조건이 true라서 9을 출력.
			//++이 되어 10이 되고, 10 < 10에서는 조건이 false가 되어 반복문을 탈출!
		{
			cout << i << endl;
		}
		//구구단 2단을 출력해보자.
		for (int i = 1; i <= 9; ++i)
		{
			cout << "2 *" << i << " = " << 2 * i << endl;
		}

		// 1 ~ 100 사이의 숫자 중 짝수만 출력하는 for문을 작성해보자. 
		for (int i = 2; i <= 100; i += 2)
		{
			cout << i << endl;
		}
		for (int i = 1; i <= 100; ++i) // 3와 7의 공배수만 출력하는 if문.
		{
			if (i % 3 == 0 && i % 7 == 0)
				cout << i << endl;
		}
		for (int i = 1; i <= 10; ++i) // 중첩 for문 : for문 안에 for문이 존재.
		{
			for (int j = 0; j < 10; ++j)
			{
				cout << "i = " << i << "j =" << j << endl;
			}
		}
	}
