	/*
	사칙연산자 : +, -, *. / , %(나머지 연산자)
	관계연산자 : 값대 갑을 연산하여 참/거짓으로 결과를 반환한다.
	>,(크다) >= , <(작다) , <=, ==(같다), !=(다르다)
	컴퓨터는 시키는 것만 하는 구조이기 때문에, 하나하나 잘할 수 있어야 한다.
	*/

	cout << "10 < 20 = " << (10 < 20) << endl;
	cout << "10 <= 20 = " << (10 <= 20) << endl;
	cout << "10 > 20 = " << (10 > 20) << endl;
	cout << "10 >= 20 = " << (10 >= 20) << endl;
	cout << "10 == 20 = " << (10 == 20) << endl;
	cout << "10 != 20 = " << (10 != 20) << endl;
	cout << " 숫자를 입력하세요 : ";
	cin >> Number;

	cout << "10  ~ 20 = " << (10 <= Number && Number <= 20) << endl; // 15, and(둘 다 맞아야 참값)
	cout << "87 & 53 = " << (87 & 53) << endl;
