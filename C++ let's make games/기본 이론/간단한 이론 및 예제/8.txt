	//8강. switch와 열거체.
	/*
	switch문 : 분기문의 한 종류이다. if문이 조건을 체크하는 분기문이라면
	switch문은 값이 뭔지 체크하는 분기문이다. 형태 : switch(변수) {}
	코드블럭 안에는 case break 구문이 들어가게 도니다.
	case 상수 의 형태로 처리가 되고 변수값이 무엇인지에 따라서 case 뒤에 들어오는 상수를 비교.
	*/
	switch (iNumber)
	{
	case 1:
		cout << "입력한 숫자는 1입니다." << endl; //break가 없을 경우 아래의 case문도 강제로 실행된다.
	case 2:
		cout << "입력한 숫자는 2입니다." << endl;
		break;
	case 3:
		cout << "입력한 숫자는 1입니다." << endl;
		break;
	case 4:
		cout << "입력한 숫자는 4입니다." << endl;
		break;
	default: //case로 지정되어 있지 않는 숫자가 들어올 경우 실행된다.
		cout << "그 외의 숫자입니다." << endl;
		break;
	}
	/*
	열거체 :연속된 숫자에 이름을 부여할 수 있는 기능이다. enum열거체명 {}의 형태로 구성된다.
	열거체명을 이용해서 열거체 타입의 변수를 선언도 가능하다. 즉, 열거체 = 사용자정 변수 타입.
	*/
	enum NUM
	{
		NUM_0, //아무것도 부여하지 않을 경우 0~1에서 차례로 값이 부여,
		NUM_1,
		NUM_2 = 0x0010, //내가 원하는 값을 지정할 수 있다.
		NUM_3
	};
#define NUM_4 4 

	enum SRP
	{
		SRP_S = 1,
		SRP_R,
		SRP_P,
		SRP_END
	};
	NUM eNum; // 열거체 타입의 변수를 선언했다. 열거체 타입 변수는 무조건 4byte를 차지하게 된다.
	//sizeof(타입 or 변수)를 하게 되면 해당 타입 또는 변수의 메모리 크기를 구한다.
	cout << sizeof(NUM) << endl;
	//typeid(타입 or 변수).name()을 하게 되면 typeid 안에 들어간 타입 혹은 변수의 타입을 문자열로 반환한다. 
	cout << typeid(eNum).name() << endl;
