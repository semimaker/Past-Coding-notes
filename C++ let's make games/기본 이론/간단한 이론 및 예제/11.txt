	for (int i = 0; i < 5; ++i) // 별 증가
	{
		for (int j = 0; j < i + 1; ++j)
		{
			cout << "*";
		}

		cout << endl;
	}

	for (int i = 0; i < 5; ++i) //별 감소
	{
		for (int j = 0; j < 5 - i; ++j)
		{
			cout << "*";
		}

		cout << endl;
	}
	//공백 : 3,2,1,0. 별 : 1,3,5,7
	for (int i = 0; i < 5; ++i) //별 감소
	{
		//공백을 출력하기 위한 for문
		for (int j = 0; j < 3 - i; ++j)
		{
			cout << " ";
		}
		//*을 출력하기 위한 for문
		for (int j = 0; j < i * 2 + 1; ++j)
		{
			cout << "*";
		}
		cout << endl;
	}
	//구구단을 2~9단으로 출력한다.
	for (int i = 2; i <= 9; ++i)
	{
		for (int j = 1; j <= 9; ++j)
		{
			cout << i << "*" << j << " = " << i * j << endl;
		}
	}
	/* 2단 3 4
	   5   6 7
	   8   9 10단 */
	for (int i = 2; i < 11; i += 3)
	{
		for (int j = 1; j <= 9; ++j)
		{
			cout << i << "*" << j << " = " << i * j << "Wt";
			cout << i + 1 << "*" << j << " = " << (i + 1) * j << "Wt";
			cout << i + 2 << "*" << j << " = " << (i + 2) * j << endl;
		}
		cout << endl;
	}
	cout << endl;

	int iLine = 7; // 11로도 해서 확인이 가능하다. 
	int iCount = 0;

	for (int i = 0; i < iLine; ++i)
	{
		//공백 : 3,2,1,0,1,2,3
		//별 : 1,3,5,7,5,3,1

		iCount = i;
		//i값이 7줄 기준 4,5,6 즉, 밑의 삼각형을 구성할 때만 if문에 들어가게 된다.
		if (i >= iLine / 2)
		{
			iCount = iLine - 1 - i;
		}

		// i값이 0,1,2,3일 때는 iCount는 i값을 그대로 대입된다.
		// i값이 4,5,6일떄는 iCount는 2,1,0이 된다.
		//즉, 최종 i값은 0,1,2,3,2,1,0으로 돌아가게 된다.
		for (int j = 0; j < iLine / 2 - iCount; ++j)
		{
			cout << "  ";
		}

		for (int j = 0; j < 3 - iCount; ++j)
		{
			cout << "  " << endl;
		}
	}
