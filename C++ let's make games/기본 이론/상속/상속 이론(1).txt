//상속 : 클래스는 서로간 부모, 자식 관계를 형성. 자식 - 부모 클래스 멤버 사용 가능.
//부모클래스에서 public, protected 로 되어 있는 멤버 사용 가능
//private 멤버는 자식에서도 사용이 안된다.

//형태 : public, private. 

class CParent
{
public:
	CParent()
	{
		cout << "CParent 생성자 " << endl;
	}

	~CParent()
	{
		cout << "CParent 소멸자 " << endl;
	}
public:
	int m_iA;
protected:
	int m_iB;
private:
	int m_iC;
public:
	void Output()
	{
		cout << "A : " << m_iA << endl;
		cout << "B : " << m_iB << endl;
		cout << "C : " << m_iC << endl;
	}
};

class CChild : public CParent
{
public:
	CChild()
	{
		cout << "Child 생성자 " << endl;
	}
	~CChild()
	{
		cout << "Child 소멸자 " << endl;
	}

protected:
	int m_iD;
};

class CChild1 : private CParent
{
public:
	CChild1()
	{	
		m_iA = 100;
		m_iB = 200;
		//m_iC = 300; private은 내부에서 접근이 안된다.
		cout << "Child1 생성자 " << endl;
	}
	~CChild1()
	{
		cout << "Child1 소멸자 " << endl;
	}
};

class CChildChild : public CChild
{
public:
	CChildChild()
	{
		m_iD = 500;
		cout << "CCildChild 생성자 " << endl;
	}
	~CChildChild()
	{
		cout << "CCildChild 소멸자 " << endl;
	}
private:
	int m_iE;
};
