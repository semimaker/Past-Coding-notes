#include <iostream>

//코드블럭 외부에 변수를 선언하면 전역변수가 된다. 전역변수는 이 변수를 선언한 아래의 모든 부분에서 사용가능하다.
//전역변수는 소속된 코드블럭이 없으므로 프로그램이 종료될때까지 메모리에 존재하며 종료 시에는 메모리에서 해제된다.
using namespace std;
/*
함수 : 기능을 만든다. 일종의 사용가능한 부품을 제작. 특정 상황에 동작하는 코드 집합을 생성.
형태 : 반환타입 함수명(인자) {} 
반환 타입 : 이 함수가 기능을 수행하고 기능의 결과를 반환 할 경우 지정. 정수를 반환할 때는 정수타입,
실수를 반환할때에는 실수타입으로 지정. 그외에는 해당타입을 지정
함수 명 = 함수 이름. 가급적 함수가 하는 역할을 정확히 표시하는 것이 가독성에 좋다.
인자 : 해당 함수를 사용할때 넘겨줄 값이 있을 경우 인자를 활용. 있기도 없기도 하다.

해당 함수를 호출하면 코드블럭 안의 코드가 동작이 된다.
*/
int g_iNum;

int Sum(int a, int b)
{
	return a + b; // 이 키워드로 함수 반환타입에 맞는 값을 계산.
}

void Output(int iNum1, int iNum2 = 10)
{
	cout << iNum1 << endl;
	cout << iNum2 << endl;
}
void OutputText()
{
	cout << "OutputText Function!" << endl;
}
// 타입이 없다. 반환 의무가 없다. 단순히 만들기 하면 동작하지 않고, 기능 동작을 위해서는 반드시
//main함수 안에서 동작시켜야 한다.
void OutputNumber(int iNumber)
{
	iNumber = 9999;
}
void ChangeNumber(int iNumber) // 바로 아래와 같은 이름의 함수이지만, 정수와 주소를 주는 차이.
{
	iNumber = 9999;
	g_iNum = 456;
}
void ChangeNumber(int* pNumber) // 결과값을 인자로 받아서 발생.(call by address)
{
	*pNumber = 2939;
}//함수의 오버로딩 : 같은 이름의 함수이지만, 기능이 다르던가, 인자가 다른 함수를 의미.
//조건 : 함수 이름이 무조건 같아야 한다. 단, 인자의 개수 혹은 타입이 달라야 오버로딩이 성립된다.
int main() {//아래 함수처럼 결과를 반환하는 형태 : call by value.

	//함수 호출 시 함수명(인자로 넘겨줄 값); 형태로 호출
	//아래처럼 호출하면, a -> 10, b -> 20 이 전달되어 더한 30이 나온다.
	cout << Sum(10, 20) << endl;
	cout << Sum(102, 304) << endl;

	OutputText();
	OutputNumber(1234);
	
	// 아래처럼 특정 블럭 안에서 선언되는 변수 : 지역변수.
	//지역변수는 메모리 영역 중 stack에 할당된다.
	int iNumber = 0;
	ChangeNumber(iNumber); //void에서 일반값과 함수값을 받는 것의 차이.
	ChangeNumber(&iNumber);//포인터를 활용한 오버플로우

	cout << iNumber <<endl;
	cout << g_iNum <<endl;

	Output(10, 30);

	{
		int iNum1 = 1234;
	}

	cout << iNumber << endl;
	//iNum1은 위의 코드블럭 안에 존재하는 변수이므로, 저 코드블럭 안에서만 사용할 수 있다.
	//cout << iNum1 << endl;
	return 0;
}
