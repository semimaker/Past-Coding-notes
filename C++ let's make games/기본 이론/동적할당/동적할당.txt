#include <iostream>

using namespace std;

int main()
{
	//동적할당 : 메모리 생성을 원하는 시점에 할 수 있는 기능.
	//메모리 영역 중 힙 영역에 공간이 할당. 힙 특성상 메모리를 해제하기 전에는 남아있다.
	
	//메모리 릭 : 동적할당을 하고 메모리를 해제하지 않으면 사용하지 않지만 공간은 계속
	// 잡혀있다. 해당 공간은 다른 곳에 사용이 불가능하다.

	//동적할당은 new 키워드를 이용하며, C언어에서는 malloc()함수를 이용한다.
	
	//new 할당할 타입; 
	//new, malloc 둘 다 마찬가지로 힙 영역에 메모리 공간을 할당하고 해당 메모리의 주소를
	//반환해준다.

	int* pNumber = new int[100]; // 동적배열 할당방법 : new 타입[개수] 형태.
	*pNumber = 100;

	//아래처럼 동적배열로 할당하면 메모리 공간에 400바이트만큼 공간을 할당하고 해당 메모리의
	//시작 주소를 반환한다. 

	delete pNumber; // 이 키워드로 동적할당한 메모리를 해제. 메모리 주소를 넣은 형태.

	//new 뒹에 들어오는 타입인 int 크기만큼 공간이 할당된다.
	//4byte 공간을 할당. 그 뒤, 할당한 공간의 메모리 주소를 pNumber에 넣어준다.

	int* pArray = new int[100];
	pArray[1] = 300;
	delete[] pArray;

	return 0;
}
