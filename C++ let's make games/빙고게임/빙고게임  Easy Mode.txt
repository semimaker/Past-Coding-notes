#include <iostream>
#include <time.h> //<stdio.h>가 아닌 <time.h>으로서 사용.

using namespace std;

enum AI_MODE
{
	AM_EASY = 1,
	AM_HARD
	//AI 난이도를 선택한다.
};
int main()
{
	srand((unsigned int)time(0));

	int iNumber[25] = {};
	int iAINumber[25] = {};


	//1 ~25  숫자 입력
	for (int i = 0; i < 25; ++i)
	{
		iNumber[i] = i + 1;
		iAINumber[i] = i + 1;
	} // --- 여기 괄호를 넣지 않아서 수가 배열되지 않음.
	//숫자를 섞어준다.
	int iTemp, idx1, idx2;
	for (int i = 0; i < 100; ++i)
	{
		idx1 = rand() % 25;
		idx2 = rand() % 25; //숫자를 랜덤으로 반환. 0 ~ RAND_MAX(32767)

		iTemp = iNumber[idx1];
		iNumber[idx1] = iNumber[idx2];
		iNumber[idx2] = iTemp;

		idx1 = rand() % 25;
		idx2 = rand() % 25; //숫자를 랜덤으로 반환. 0 ~ RAND_MAX(32767)

		iTemp = iNumber[idx1];
		iAINumber[idx1] = iAINumber[idx2];
		iAINumber[idx2] = iTemp;
	}

	int iAIMode;
	int iBingo = 0 ,iAIBingo = 0;

	while (true)
	{
		system("cls");
		cout << "1. Easy" << endl;
		cout << "2. Hard" << endl;//AI 난이도를 선택한다.
		cout << "AI 모드를 선택하세요. : ";
		cin >> iAIMode;

		if (iAIMode >= AM_EASY && iAIMode <= AM_HARD)
			break;
	}

	/*
	AI Easy 모드는 현재 AI 숫자목록 중 *로 바뀌지 않은 숫자 중에서 그 목록 중 하나를 선택하게 된다.(랜덤하게)
	*/
	//선택이 되지 않은 목록 배열을 만들어준다.
	int iNoneSelect[25] = {};
	//선택 안된 숫자 개수를 저장한다
	int iNoneSelectCount = 0;

	while (true)
	{
		system("cls");

		//숫자는 5 x 5으로 출력.
		cout << "=========== Player ==============" << endl;

		for (int i = 0; i < 5; ++i)
		{
			for (int j = 0; j < 5; ++j)
			{
				//if (iNumber[i * 5 + j] == INT_MAX)
				//	cout << "*\t";
				//else
				cout << iNumber[i * 5 + j] << "\t";
			}
			cout << endl;
		} //지우고 다시 출력하는 과정이 반복된다.

		cout << "Bingo Line : " << iBingo << endl; //줄수가 5 이상일 경우, 게임을 종료한다. 

		cout << "=========== AI ==============" << endl;
		
		switch (iAIMode)
		{
		case AM_EASY:
			cout << "AIMode : Easy" << endl;
			break;
		case AM_HARD:
			cout << "AIMode : Hard" << endl;
			break;
		}

		for (int i = 0; i < 5; ++i)
		{
			for (int j = 0; j < 5; ++j)
			{
				if (iAINumber[i * 5 + j] == INT_MAX)
					cout << "*\t";
				else
					cout << iAINumber[i * 5 + j] << "\t";
			}

			cout << endl;
		}

		cout << "AIBingo Line : " << iAIBingo << endl;

		if (iBingo >= 5)
		{
			cout << "Player 승리" << endl;
			break;
		}
		else if (iAIBingo >= 5)
		{
			cout << "AI 승리" << endl;
			break;
		}

		cout << "숫자를 입력하세요(0 : 종료) : ";
		int iInput;
		cin >> iInput;// 예외처리 - 허용되지 않은 상황에 대해 처리하는 것.

		if (iInput == 0)
			break;
		else if (iInput < 1 || iInput > 25)
			continue;

		bool bAcc = true; //중복변수를 체크하는 변수. 중복시 true로 명명.
		//모든 숫자를 차례대로 검사해서 입력한 숫자와 같은 숫자가 있는지를 찾아낸다.

		for (int i = 0; i < 25; ++i)
		{
			if (iInput == iNumber[i]) // 같은 숫자가 있는 경우
			{
				bAcc = false; //숫자를 찾으면 중복된 숫자가 아니므로 bAcc 변수를 false로.

				iNumber[i] = INT_MAX;
				//더이상 다른 숫자를 찾을 필요가 없으므로 for문을 빠져나간다.
				break;
			}
		}

		if (bAcc)
			continue;// 빙고 줄 수를 체크하는 것은 매번 입력할 떄마다 처음부터 세로 카운트

		for (int i = 0; i < 25; ++i)
		{
			if (iAINumber[i] == iInput)
			{
				iAINumber[i] = INT_MAX;
				//플레이어가 입력한 숫자를 찾아서 *로 바꾸었으므로, 선택 안된 숫자가 1개 줄어든다.

				--iNoneSelectCount;
				break;
			}
		}


		switch (iAIMode)
		{
			
		case AM_EASY: //선택안된 숫자목록을 만들어준다. 선택 안된 숫자 개수는 목록을 만들 떄 카운팅해준다.
			
			iNoneSelectCount = 0;
			for (int i = 0; i < 25; ++i)
			{
				if (iAINumber[i] != INT_MAX) // 현재 숫자가 *가 아닐 경우.
				{
					iNoneSelect[iNoneSelectCount] = iAINumber[i];
					++iNoneSelectCount;
					//*이 아닌 숫자일 경우, iNoneSelectCount을 인덱스로 활용.
					//선택 안된 목록에 추가할 때마다 개수를 1씩 증가.
					//이로 인하여 총 선택안된 개수를 구해준다.
					//그런데 0부터 카운트가 시작이므로, 0번에 넣고 ++해서 1개 추가
					//되었다고 해준다.
				}
			}

			//for문을 빠져나오게 되면, 선택 안된 목록이 만들어지고 선택안된
			//목록의 개수가 만들어지게 된다.
			//선택안된 목록의 숫자 중 랜덤한 하나의 숫자를 얻어오기 위해 
			//인덱스를 랜덤하게 구해준다.
			iInput = iNoneSelect[rand() % iNoneSelectCount];
			break;

		case AM_HARD:
			break;

		}

		for (int i = 0; i < 25; ++i)
		{
			if (iNumber[i] = iInput)
			{
				iNumber[i] = INT_MAX;
				break;
			}
		}

		for (int i = 0; i < 25; ++i)
		{
			if (iAINumber[i] = iInput)
			{
				iAINumber[i] = INT_MAX;
				break;
			}
		}
		//AI가 숫자를 선택했으므로, 플레이어와 AI 숫자를 *로 바꾸어준다.
		//중복이 아니라면 AI 숫자도 찾아서 *로 바뀌준다.

		iBingo = 0;//그러므로 iBingo 변수를 0으로 매번 초기화하고 새롭게 줄 수를 구해야 한다.
		iAIBingo = 0;

		int iStar1 = 0, iStar2 = 0;
		int iAIStar1 = 0, iAIStar2 = 0;
		for (int i = 0; i < 5; ++i)
		{
			for (int j = 0; j < 5; ++j)
			{
				//한줄을 체크하기 전에 먼저 0으로 별 개수를 초기화.
				iStar1 = iStar2 = 0;
				iAIStar1 = iAIStar2 = 0;
				for (int j = 0; j < 5; ++j)
				{
					if (iNumber[i * 5 + j] == INT_MAX) //가로 별 개수를 구해준다.
						++iStar1;
					
					if (iNumber[j * 5 + i] == INT_MAX) //세로 별 개수를 구해준다.
						++iStar2;
					
					if (iAINumber[i * 5 + j] == INT_MAX) //가로 별 개수를 구해준다.
						++iAIStar1;
					
					if (iAINumber[j * 5 + i] == INT_MAX) //세로 별 개수를 구해준다.
						++iAIStar2;

				} //for문을 빠져나오면서 현재 줄의 가로 별 개수가 몇개인지 iStart1 변수에 들어가게 된다.
				//iStar1의 값이 5이면, 한줄이 모두 *이므로 줄 카운트 추가.
				if (iStar1 == 5)
					++iBingo;
				
				if (iStar2 == 5)
					++iBingo;// 오른쪽 상단.
				
				if (iAIStar1 == 5)
					++iAIBingo;
				
				if (iAIStar2 == 5)
					++iAIBingo;// 오른쪽 상단.
			}
			//왼쪽 상단 -> 오른쪽 하단 대각선 체크
			
			iStar1 = 0;
			iAIStar1 = 0;
			for (int i = 0; i < 25; i += 6)
			{
				if (iNumber[i] == INT_MAX) //가로 별 개수를 구해준다.
					++iStar1;

				if (iAINumber[i] == INT_MAX) //가로 별 개수를 구해준다.
					++iAIStar1;
			}

			if (iStar1 == 5)
				++iBingo;

			if (iAIStar1 == 5)
				++iAIBingo;
			
			// 오른쪽 상단 -> 왼쪽 하단 대각선 체크


			iStar1 = 0;
			iAIStar1 = 0;
			for (int i = 4; i <= 20; i += 4)
			{
				if (iNumber[i] == INT_MAX)
					++iStar1;

				if (iAINumber[i] == INT_MAX)
					++iAIStar1;
			}

			if (iStar1 == 5)
				++iBingo;
			if (iAIStar1 == 5)
				++iAIBingo;
		
		} //지우고 다시 출력하는 과정이 반복된다.
	

	}// 하나의 과정이 만들어졌으므로 Test을 해볼 것.

	return 0;
}
