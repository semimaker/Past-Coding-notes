#include <iostream>
#include <time.h> //<stdio.h>가 아닌 <time.h>으로서 사용.

using namespace std;

int main()
{
	srand((unsigned int)time(0));

	int iNumber[25] = {};

	//1 ~25  숫자 입력
	for (int i = 0; i < 25; ++i)
	{
		iNumber[i] = i + 1;
	} // --- 여기 괄호를 넣지 않아서 수가 배열되지 않음.
	
	 //숫자를 섞어준다.
	int iTemp, idx1, idx2;
	for (int i = 0; i < 100; ++i)
	{
		idx1 = rand() % 25;
		idx2 = rand() % 25; //숫자를 랜덤으로 반환. 0 ~ RAND_MAX(32767)
		iTemp = iNumber[idx1];
		iNumber[idx1] = iNumber[idx2];
		iNumber[idx2] = iTemp;
	}

	int iBingo = 0;

	while (true)
	{
		system("cls");

		//숫자는 5 x 5으로 출력.
		for (int i = 0; i < 5; ++i)
		{
			for (int j = 0; j < 5; ++j)
			{
				if (iNumber[i * 5 + j] == INT_MAX)
					cout << "*\t";
				else
					cout << iNumber[i * 5 + j] << "\t";
			}
			cout << endl;
		} //지우고 다시 출력하는 과정이 반복된다.
		

		cout << "Bingo Line : " << iBingo << endl; 

		if (iBingo >= 5)
			break;

		cout << "숫자를 입력하세요(0 : 종료) : ";
		int iInput;
		cin >> iInput;// 예외처리 - 허용되지 않은 상황에 대해 처리하는 것.

		if (iInput == 0)
			break;
		else if (iInput < 1 || iInput > 25)
			continue;

		bool bAcc = true; //중복변수를 체크하는 변수. 중복시 true로 명명.
		//모든 숫자를 차례대로 검사해서 입력한 숫자와 같은 숫자가 있는지를 찾아낸다.

		for (int i = 0; i < 25; ++i)
		{
			if (iInput == iNumber[i]) // 같은 숫자가 있는 경우
			{
				bAcc = false; //숫자를 찾으면 중복된 숫자가 아니므로 bAcc 변수를 false로.

				iNumber[i] = INT_MAX;
				//더이상 다른 숫자를 찾을 필요가 없으므로 for문을 빠져나간다.
				break;
			}
		}

		if (bAcc)
			continue;

		iBingo = 0;

		int iStar1 = 0, iStar2 = 0;
		for (int i = 0; i < 5; ++i)
		{
			iStar1 = iStar2 = 0;
			for (int j = 0; j < 5; ++j)
			{
				if (iNumber[i * 5 + j] == INT_MAX)
					++iStar1;
				
				if (iNumber[j * 5 + i] == INT_MAX)
					++iStar2;
			}

			if (iStar1 == 5)
				++iBingo; //7분
			if (iStar2 == 5)
				++iBingo; //7분
		}

		iStar1 = 0;
		for (int i = 0; i < 25; i += 6)
		{
			if (iNumber[i] == INT_MAX)
				++iStar1;
		}//왼쪽 상단 -> 오른쪽 하단 대각선 체크

		if (iStar1 == 5)
			++iBingo;
		
		iStar1 = 0;
		for (int i = 4; i < 20; i += 4)
		{
			if (iNumber[i] == INT_MAX)
				++iStar1;
		}//오른쪽 상단 -> 왼쪽 하단 대각선 체크


		if (iStar1 == 5)
			++iBingo;
	}// 하나의 과정이 만들어졌으므로 Test을 해볼 것.
	
	return 0;
}
//프로그래밍은 남이 하는대로가 꼭 맞는 것은 아니다. 프로그래밍 = 창의성.
