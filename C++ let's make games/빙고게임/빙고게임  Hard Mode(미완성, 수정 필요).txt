#include <iostream>
#include <time.h> //<stdio.h>가 아닌 <time.h>으로서 사용.

using namespace std;

enum AI_MODE
{
	AM_EASY = 1,
	AM_HARD
	//AI 난이도를 선택한다.
};

enum LINE_NUMBER
{
	LN_H1,
	LN_H2,
	LN_H3,
	LN_H4,
	LN_H5,
	LN_V1,
	LN_V2,
	LN_V3,
	LN_V4,
	LN_V5,
	LN_LT,
	LN_RT
};
int main()
{
	srand((unsigned int)time(0));

	int iNumber[25] = {};
	int iAINumber[25] = {};


	//1 ~25  숫자 입력
	for (int i = 0; i < 25; ++i)
	{
		iNumber[i] = i + 1;
		iAINumber[i] = i + 1;
	} // --- 여기 괄호를 넣지 않아서 수가 배열되지 않음.
	//숫자를 섞어준다.
	int iTemp, idx1, idx2;
	for (int i = 0; i < 100; ++i)
	{
		idx1 = rand() % 25;
		idx2 = rand() % 25; //숫자를 랜덤으로 반환. 0 ~ RAND_MAX(32767)

		iTemp = iNumber[idx1];
		iNumber[idx1] = iNumber[idx2];
		iNumber[idx2] = iTemp;

		idx1 = rand() % 25;
		idx2 = rand() % 25; //숫자를 랜덤으로 반환. 0 ~ RAND_MAX(32767)

		iTemp = iNumber[idx1];
		iAINumber[idx1] = iAINumber[idx2];
		iAINumber[idx2] = iTemp;
	}

	int iAIMode;
	int iBingo = 0, iAIBingo = 0;

	while (true)
	{
		system("cls");
		cout << "1. Easy" << endl;
		cout << "2. Hard" << endl;//AI 난이도를 선택한다.
		cout << "AI 모드를 선택하세요. : ";
		cin >> iAIMode;

		if (iAIMode >= AM_EASY && iAIMode <= AM_HARD)
			break;
	}

	/*
	AI Easy 모드는 현재 AI 숫자목록 중 *로 바뀌지 않은 숫자 중에서 그 목록 중 하나를 선택하게 된다.(랜덤하게)
	*/
	//선택이 되지 않은 목록 배열을 만들어준다.
	int iNoneSelect[25] = {};
	//선택 안된 숫자 개수를 저장한다
	int iNoneSelectCount = 0;

	while (true)
	{
		system("cls");

		//숫자는 5 x 5으로 출력.
		cout << "=========== Player ==============" << endl;

		for (int i = 0; i < 5; ++i)
		{
			for (int j = 0; j < 5; ++j)
			{
				if (iNumber[i * 5 + j] == INT_MAX)
					cout << "*\t";
				else
				cout << iNumber[i * 5 + j] << "\t";
			}
			cout << endl;
		} //지우고 다시 출력하는 과정이 반복된다.

		cout << "Bingo Line : " << iBingo << endl; //줄수가 5 이상일 경우, 게임을 종료한다. 

		cout << "=========== AI ==============" << endl;

		switch (iAIMode)
		{
		case AM_EASY:
			cout << "AIMode : Easy" << endl;
			break;
		case AM_HARD:
			cout << "AIMode : Hard" << endl;
			break;
		}

		for (int i = 0; i < 5; ++i)
		{
			for (int j = 0; j < 5; ++j)
			{
				if (iAINumber[i * 5 + j] == INT_MAX)
					cout << "*\t";
				else
					cout << iAINumber[i * 5 + j] << "\t";
			}

			cout << endl;
		}

		cout << "AIBingo Line : " << iAIBingo << endl;

		if (iBingo >= 5)
		{
			cout << "Player 승리" << endl;
			break;
		}
		else if (iAIBingo >= 5)
		{
			cout << "AI 승리" << endl;
			break;
		}

		cout << "숫자를 입력하세요(0 : 종료) : ";
		int iInput;
		cin >> iInput;// 예외처리 - 허용되지 않은 상황에 대해 처리하는 것.

		if (iInput == 0)
			break;
		else if (iInput < 1 || iInput > 25)
			continue;

		bool bAcc = true; //중복변수를 체크하는 변수. 중복시 true로 명명.
		//모든 숫자를 차례대로 검사해서 입력한 숫자와 같은 숫자가 있는지를 찾아낸다.

		for (int i = 0; i < 25; ++i)
		{
			if (iInput == iNumber[i]) // 같은 숫자가 있는 경우
			{
				bAcc = false; //숫자를 찾으면 중복된 숫자가 아니므로 bAcc 변수를 false로.

				iNumber[i] = INT_MAX;
				//더이상 다른 숫자를 찾을 필요가 없으므로 for문을 빠져나간다.
				break;
			}
		}

		if (bAcc)
			continue;// 빙고 줄 수를 체크하는 것은 매번 입력할 떄마다 처음부터 세로 카운트

		for (int i = 0; i < 25; ++i)
		{
			if (iAINumber[i] == iInput)
			{
				iAINumber[i] = INT_MAX;
				//플레이어가 입력한 숫자를 찾아서 *로 바꾸었으므로, 선택 안된 숫자가 1개 줄어든다.

				--iNoneSelectCount;
				break;
			}
		}


		switch (iAIMode)
		{

		case AM_EASY: //선택안된 숫자목록을 만들어준다. 선택 안된 숫자 개수는 목록을 만들 떄 카운팅해준다.

			iNoneSelectCount = 0;
			for (int i = 0; i < 25; ++i)
			{
				if (iAINumber[i] != INT_MAX) // 현재 숫자가 *가 아닐 경우.
				{
					iNoneSelect[iNoneSelectCount] = iAINumber[i];
					++iNoneSelectCount;
					//*이 아닌 숫자일 경우, iNoneSelectCount을 인덱스로 활용.
					//선택 안된 목록에 추가할 때마다 개수를 1씩 증가.
					//이로 인하여 총 선택안된 개수를 구해준다.
					//그런데 0부터 카운트가 시작이므로, 0번에 넣고 ++해서 1개 추가
					//되었다고 해준다.
				}
			}

			//for문을 빠져나오게 되면, 선택 안된 목록이 만들어지고 선택안된
			//목록의 개수가 만들어지게 된다.
			//선택안된 목록의 숫자 중 랜덤한 하나의 숫자를 얻어오기 위해 
			//인덱스를 랜덤하게 구해준다.
			iInput = iNoneSelect[rand() % iNoneSelectCount];
			break;

		case AM_HARD:
			int iLine = 0;
			int iStarCount = 0;
			int iSaveCount = 0;

			for (int i = 0; i < 5; ++i)//가로,세로 라인 중 가장 *이 많은 라인을 찾아낸다.
			{
				iSaveCount = 0;
				for (int j = 0; j < 5; ++j)
				{
					if (iAINumber[i * 5 + j] == INT_MAX)
						++iStarCount;
				}

				if (iStarCount < 5 && iSaveCount < iStarCount)
				{
					iLine = i;
					//여기는 가로 라인 중 가장 별이 많은 라인을 체크, 가로라인은 0 ~ 4로 의미를 부여했다.
					iSaveCount = iStarCount;
					//별이 5개 미만이어야 빙고 줄이 아니고 기존에 가장 많은
					//라인의 별보다 커야 가장 별이 많은 라인이므로, 라인을
					//교체해주고 저장된 별 수를 교체한다.
				}
			}

			for (int i = 0; i < 5; ++i)
			{
				iSaveCount = 0;
				for (int j = 0; j < 5; ++j)
				{
					if (iAINumber[j * 5 + i] == INT_MAX)
						++iStarCount;
				}

				if (iStarCount < 5 && iSaveCount < iStarCount)
				{
					//세로 라인은 5 ~ 9로 의미를 부여했다.
					iLine = i + 5;
					iSaveCount = iStarCount;
				}
				iStarCount = 0; // 왼쪽 -> 오른쪽 대각선 체크.
				for (int i = 0; i < 25; i += 6)
				{
					if (iAINumber[i] == INT_MAX)
						++iStarCount;
				}

				if (iStarCount < 5 && iSaveCount < iStarCount)
				{
					iLine = LN_LT;
					iSaveCount = iStarCount;
				}

				iStarCount = 0;
				for (int i = 4; i <= 20; i += 4)
				{
					if (iAINumber[i] == INT_MAX)
						++iStarCount;
				}

				if (iStarCount < 5 && iSaveCount < iStarCount)
				{
					iLine = LN_RT;
					iSaveCount = iStarCount;
				}

				if (iLine <= LN_H5)
				{
					for (int i = 0; i < 5; ++i)
					{
						if (iAINumber[iLine * 5 + i] != INT_MAX)
						{
							iInput = iAINumber[iLine * 5 + i];
							break;
						}
					}
				}

				else if (iLine <= LN_V5)
				{
					//세로줄일 경우 iLine이 5 ~ 9 사이의 값이다.
					for (int i = 0; i < 5; ++i)
					{
						if (iAINumber[i * 5 + (iLine - 5)] != INT_MAX)
						{
							iInput = iAINumber[i * 5 + (iLine - 5)];
							break;
						}
					}
				}


				else if (iLine <= LN_LT)
				{
					for (int i = 0; i < 25; i += 6)
					{
						if (iAINumber[i] != INT_MAX)
						{
							iInput = iAINumber[i];
							break;
						}
					}
				}


				else if (iLine <= LN_RT)
				{
					for (int i = 4; i < 20; i += 4)
					{
						if (iAINumber[i] != INT_MAX)
						{
							iInput = iAINumber[i];
							break;
						}
					}
				}
				
				break;
			}
			//가로 라인 중 가장 별이 많은 라인은 이미 구했다.
			//이 값과 세로 라인들을 비교하여 별이 가장 많은 라인을 구한다.

			//하드모드는 현재 숫자 중 빙고줄 완성 가능성이 가장 높은 줄을 찾아서
			//찾아서 그 줄에 있는 숫자 중 하나늘 *로 만들어준다.	
		}

	}
}
//프로그래밍은 남이 하는대로가 꼭 맞는 것은 아니다. 프로그래밍 = 창의성.
